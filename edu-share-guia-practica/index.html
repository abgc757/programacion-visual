    def alternar_cambio(self):
        # «A cambio de» solo se puede escribir si el tipo es Intercambio
        if self.v_tipo.get() == "Intercambio":
            self.e_cambio.state(["!disabled"])
        else:
            self.v_cambio.set("")
            self.e_cambio.state(["disabled"])
    def validar(self):
        # Devuelve (mensaje, campo) si hay un problema, o None si todo está bien
        if len(self.v_nombre.get().strip()) < 3:
            return "Escribe el nombre del artículo (mínimo 3 letras).", self.e_nombre
        if not self.v_categoria.get():
            return "Elige una categoría.", self.c_categoria
        if self.v_tipo.get() == "Intercambio" and not self.v_cambio.get().strip():
            return "Escribe qué te gustaría recibir a cambio.", self.e_cambio
        if not self.v_contacto.get().strip():
            return "Escribe un correo o teléfono para que te contacten.", self.e_contacto
        if not self.v_acepto.get():
            return "Debes aceptar entregar el artículo dentro del plantel.", self.e_nombre
        return None
    def publicar(self):
        problema = self.validar()
        if problema:
            mensaje, campo = problema
            messagebox.showwarning("Falta información", mensaje)
            campo.focus_set()
            return
        articulo = {
            "id": self.siguiente_id,
            "nombre": self.v_nombre.get().strip(),
            "categoria": self.v_categoria.get(),
            "estado": self.v_estado.get(),
            "tipo": self.v_tipo.get(),
            "a_cambio": self.v_cambio.get().strip(),
            "descripcion": self.t_descripcion.get("1.0", "end").strip(),
            "contacto": self.v_contacto.get().strip(),
            "situacion": "Disponible",
            "fecha": date.today().strftime("%d/%m/%Y"),
        }
        self.articulos.append(articulo)
        self.siguiente_id += 1
        self.guardar()
        self.refrescar_catalogo()
        self.limpiar_formulario()
        self.estado.set(f"Publicaste «{articulo['nombre']}»")
        messagebox.showinfo("¡Publicado!", f"«{articulo['nombre']}» ya está en el catálogo. ¡Gracias por compartir!")
        self.nb.select(self.tab_catalogo)
    def limpiar_formulario(self):
        self.v_nombre.set("")
        self.v_categoria.set("")
        self.v_estado.set("Buen estado")
        self.v_tipo.set("Donación")
        self.v_contacto.set("")
        self.v_acepto.set(False)
        self.t_descripcion.delete("1.0", "end")
        self.alternar_cambio()
        self.e_nombre.focus_set()

    def refrescar_catalogo(self):
        texto = self.v_buscar.get().strip().lower()
        categoria = self.v_filtro_cat.get()
        seleccionado = self.tabla.selection()   # recordamos qué fila estaba elegida
        for fila in self.tabla.get_children():
            self.tabla.delete(fila)
        for a in self.articulos:
            # Cada «continue» salta al siguiente artículo si no cumple el filtro
            if texto and texto not in a["nombre"].lower() and texto not in a["descripcion"].lower():
                continue
            if categoria != "Todas" and a["categoria"] != categoria:
                continue
            if self.v_solo_disp.get() and a["situacion"] != "Disponible":
                continue
            self.tabla.insert("", "end", iid=str(a["id"]),
                              values=(a["nombre"], a["categoria"], a["tipo"], a["estado"], a["situacion"]),
                              tags=(a["situacion"].lower(),))
        if seleccionado and self.tabla.exists(seleccionado[0]):
            self.tabla.selection_set(seleccionado[0])   # la volvemos a elegir
        self.actualizar_resumen()
    def articulo_seleccionado(self):
        seleccion = self.tabla.selection()
        if not seleccion:
            messagebox.showinfo("Elige un artículo", "Primero haz clic en un artículo de la lista.")
            return None
        id_buscado = int(seleccion[0])
        for a in self.articulos:
            if a["id"] == id_buscado:
                return a
        return None
    def ver_detalle(self, evento=None):
        a = self.articulo_seleccionado()
        if a is None:
            return
        ventana = tk.Toplevel(self)
        ventana.title("Detalle del artículo")
        ventana.geometry("480x400")
        ventana.transient(self)

        marco.pack(fill="both", expand=True)
        ttk.Label(marco, text=a["nombre"], style="Titulo.TLabel", wraplength=430).pack(anchor="w", pady=(0, 8))
        lineas = [("Categoría", a["categoria"]), ("Estado", a["estado"]), ("Tipo", a["tipo"])]
        if a["tipo"] == "Intercambio":
            lineas.append(("A cambio de", a["a_cambio"]))
        lineas += [("Situación", a["situacion"]), ("Publicado", a["fecha"]), ("Contacto", a["contacto"])]
        for etiqueta, valor in lineas:

        ttk.Label(marco, text=a["descripcion"] or "Sin descripción", wraplength=430,
                  relief="groove", padding=8).pack(fill="x", pady=12)
        ttk.Button(marco, text="Cerrar", command=ventana.destroy).pack(anchor="e")
    def me_interesa(self):
        a = self.articulo_seleccionado()
        if a is None:
            return
        if a["situacion"] != "Disponible":
            messagebox.showinfo("No disponible", f"Este artículo ya está {a['situacion'].lower()}.")
            return
        if messagebox.askyesno("Me interesa", f"¿Quieres reservar «{a['nombre']}»?\n\nContacto: {a['contacto']}"):

    def guardar(self):
        try:
            with open(ARCHIVO, "w", encoding="utf-8") as archivo:
                json.dump(self.articulos, archivo, ensure_ascii=False, indent=2)
        except OSError as error:
            messagebox.showerror("No se pudo guardar", str(error))
    def cargar(self):
        if not os.path.exists(ARCHIVO):
            return
        try:
            with open(ARCHIVO, encoding="utf-8") as archivo:
                self.articulos = json.load(archivo)
        except (OSError, json.JSONDecodeError):
            messagebox.showwarning("Archivo dañado", "No se pudo leer articulos.json; se empieza con la lista vacía.")
            self.articulos = []
        # El siguiente id es el más grande que existe + 1
        self.siguiente_id = max([a["id"] for a in self.articulos], default=0) + 1
    def cerrar(self):
        self.guardar()
        self.destroy()

    def actualizar_resumen(self):
        total = len(self.articulos)
        disponibles = sum(1 for a in self.articulos if a["situacion"] == "Disponible")
        entregados = sum(1 for a in self.articulos if a["situacion"] == "Entregado")
        donaciones = sum(1 for a in self.articulos if a["tipo"] == "Donación")
        intercambios = total - donaciones
        self.estado.set(f"Artículos: {total} · Disponibles: {disponibles} · Entregados: {entregados} · "
                        f"Donaciones: {donaciones} · Intercambios: {intercambios}")

        self.tabla.focus_set()
    def ayuda(self, evento=None):
        messagebox.showinfo("Atajos de teclado",
                            "Ctrl + N → publicar un artículo\n"
                            "Ctrl + B → ver el catálogo\n"
                            "Enter (en la tabla) → ver detalle\n"
                            "F1 → esta ayuda\n\n"
                            "Tab → pasar al siguiente campo")
    def atajos(self):
        self.bind("<Control-n>", self.ir_a_publicar)
        self.bind("<Control-b>", self.ir_a_catalogo)
        self.bind("<F1>", self.ayuda)
        self.tabla.bind("<Return>", self.ver_detalle)
    def crear_accesibilidad(self):
        grupo = ttk.Frame(self.cabecera)
        grupo.pack(side="right")
        ttk.Label(grupo, text="Tamaño de letra:").pack(side="left", padx=(0, 6))
        self.v_tam = tk.IntVar(value=11)
        ttk.Spinbox(grupo, from_=10, to=16, width=4, textvariable=self.v_tam,
                    command=self.aplicar_tamano).pack(side="left")
        ttk.Button(grupo, text="? Ayuda", command=self.ayuda).pack(side="left", padx=(10, 0))
    def aplicar_tamano(self):
        tam = self.v_tam.get()
        estilo = ttk.Style(self)
        estilo.configure(".", font=("Arial", tam))
        estilo.configure("Treeview", rowheight=tam * 2 + 6, font=("Arial", tam))
        estilo.configure("Treeview.Heading", font=("Arial", tam, "bold"))
        self.t_descripcion.configure(font=("Arial", tam))
✔ Así sabes que funciona: Ctrl+N, Ctrl+B y F1 funcionan; el tamaño de letra cambia.
