Paso 3Botón y tabla

def registrar():
    nombre = e_nombre.get().strip()
    if nombre == "":
        messagebox.showwarning("Falta dato", "Escribe el nombre del visitante")
        return
    cita = "Sí" if v_cita.get() else "No"
    tabla.insert("", "end", values=(nombre, c_motivo.get(), cita))
    e_nombre.delete(0, tk.END)
    v_cita.set(False)

tk.Button(ventana, text="Registrar visita", command=registrar).pack(pady=6)

tabla = ttk.Treeview(ventana, columns=("n", "m", "c"), show="headings", height=9)
tabla.heading("n", text="Nombre")
tabla.heading("m", text="Motivo")
tabla.heading("c", text="Cita")
tabla.column("c", width=60)
tabla.pack(padx=10, pady=6, fill="x")

Paso 4Eliminar y cerrar

def eliminar():
    sel = tabla.selection()
    if not sel:
        messagebox.showinfo("Aviso", "Selecciona una fila")
        return
    tabla.delete(sel[0])

tk.Button(ventana, text="Eliminar seleccionada", command=eliminar).pack()

ventana.mainloop()
